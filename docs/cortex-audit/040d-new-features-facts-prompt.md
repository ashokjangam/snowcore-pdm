# Cortex prompt 040d — read-only facts for the new features

Use connection `hackathon`. Read-only: SELECT only. Print every result as a
markdown table with ALL rows and ALL columns verbatim. Do not summarise.

```sql
SELECT SCOPE, AUTO_WORK_ORDERS, CONFIRMED_WITHIN_4H, ROUND(HIT_RATE,4) HIT_RATE,
       ROUND(RANDOM_HOUR_HIT_RATE,4) RANDOM_RATE, ROUND(LIFT_VS_RANDOM,3) LIFT,
       MEDIAN_LEAD_TIME_MIN, ORDERS_WITH_KIT_RISK, BREAKDOWNS_IN_PERIOD,
       BREAKDOWNS_COVERED, ROUND(COVERAGE_RATE,4) COVERAGE_RATE
FROM SNOWCORE_REAL.GOLD.V_RISK_WORK_ORDER_SUMMARY ORDER BY SCOPE;

SELECT MACHINE_CODE, ALARM_CODE, LONG_BREAKDOWNS, ROUND(BREAKDOWN_HOURS,1) HOURS,
       ROUND(SHARE_OF_LINE_BREAKDOWN_PCT,1) SHARE_PCT, ROUND(MEDIAN_STOP_MIN,1) MEDIAN_MIN,
       ROUND(PRECURSOR_RATE,3) PRECURSOR_RATE, ROUND(REPEAT_24H_RATE,3) REPEAT_24H_RATE,
       ROUND(EARLY_SHARE,3) EARLY, ROUND(LATE_SHARE,3) LATE, ROUND(NIGHT_SHARE,3) NIGHT,
       ROUND(BACK_TO_RUNNING_RATE,3) BACK_TO_RUN, ROUND(WAITING_AFTER_RATE,3) WAITING_AFTER,
       ROUND(ANOTHER_BREAKDOWN_RATE,3) ANOTHER_BD, PATTERN
FROM SNOWCORE_REAL.GOLD.V_ROOT_CAUSE_ALARM WHERE RANK_IN_LINE <= 3
ORDER BY MACHINE_CODE, RANK_IN_LINE;

SELECT PATTERN, COUNT(*) CODES, SUM(LONG_BREAKDOWNS) BREAKDOWNS
FROM SNOWCORE_REAL.GOLD.V_ROOT_CAUSE_ALARM GROUP BY PATTERN ORDER BY BREAKDOWNS DESC;

SELECT OUTCOME, STOCK_STATE, COUNT(*) N
FROM SNOWCORE_REAL.GOLD.RISK_WORK_ORDER GROUP BY 1,2 ORDER BY 1,2;
```
