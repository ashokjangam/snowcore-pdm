# Cortex prompt 039 — read-only facts for the dataset report

Use connection `hackathon`. Read only. Do not create, alter, insert, update,
delete or CALL anything. Prefix every statement with the database context
because tool calls do not keep session state.

Run these queries and return every result as a plain markdown table with the
exact numbers (no rounding beyond what the SQL does, no commentary).

```sql
-- A. Interval states: how much of the source each state covers
SELECT STATE_TYPE,
       COUNT(*) AS INTERVALS,
       COUNT_IF(ALARM_CODE IS NOT NULL AND ALARM_CODE <> '') AS WITH_ALARM,
       ROUND(SUM(ELAPSED_MS)/3600000,1) AS HOURS
FROM SNOWCORE_REAL.BRONZE.PIADE_INTERVALS_RAW
GROUP BY STATE_TYPE ORDER BY HOURS DESC;

-- B. Coverage window and line count
SELECT MIN(INTERVAL_START) AS FIRST_TS, MAX(INTERVAL_START) AS LAST_TS,
       COUNT(DISTINCT EQUIPMENT_ID) AS LINES, COUNT(*) AS INTERVAL_ROWS
FROM SNOWCORE_REAL.BRONZE.PIADE_INTERVALS_RAW;

-- C. Per line OEE and loss hours
SELECT MACHINE_CODE, ROUND(OEE,4) OEE, ROUND(AVAILABILITY,4) A,
       ROUND(PERFORMANCE,4) P, ROUND(QUALITY,4) Q,
       ROUND(BREAKDOWN_HOURS,1) BREAKDOWN_H, ROUND(IDLE_HOURS,1) IDLE_H,
       ROUND(ACTUAL_OUTPUT_UNITS,0) PACKAGES_OUT, WORK_ORDER_COUNT,
       ROUND(LAST_RISK_SCORE,4) LAST_RISK, RISK_BAND, MODEL_VS_BASELINE_TRUST
FROM SNOWCORE_REAL.GOLD.V_EXECUTIVE_FLEET ORDER BY MACHINE_CODE;

-- D. Site KPI
SELECT ROUND(SITE_WEIGHTED_OEE,4) OEE, ROUND(AVAILABILITY,4) A,
       ROUND(PERFORMANCE,4) P, ROUND(QUALITY,4) Q,
       ROUND(BREAKDOWN_HOURS,1) BREAKDOWN_H, ROUND(IDLE_HOURS,1) IDLE_H,
       WORK_ORDER_COUNT, STOCK_RISK_COUNT,
       ROUND(ESTIMATED_MARGIN_EXPOSURE_EUR,0) EXPOSURE_EUR
FROM SNOWCORE_REAL.GOLD.V_EXECUTIVE_KPI;

-- E. Top 8 breakdown alarm codes by stop hours
SELECT CAUSE_CODE, COUNT(*) ORDERS,
       ROUND(SUM(SOURCE_EVENT_MINUTES)/60,1) STOP_HOURS
FROM SNOWCORE_REAL.GOLD.WORK_ORDER WHERE PLANT_CODE='PLANT_B'
GROUP BY CAUSE_CODE ORDER BY STOP_HOURS DESC LIMIT 8;

-- F. Maintenance cost versus waiting cost
SELECT 'BREAKDOWN' KIND, COUNT(*) EVENTS,
       ROUND(SUM(SOURCE_EVENT_MINUTES)/60,1) HOURS,
       ROUND(SUM(LABOUR_COST),0) LABOUR_EUR, ROUND(SUM(PARTS_COST),0) PARTS_EUR,
       ROUND(SUM(LOST_PRODUCTION_COST),0) FORGONE_EUR
FROM SNOWCORE_REAL.GOLD.WORK_ORDER WHERE PLANT_CODE='PLANT_B'
UNION ALL
SELECT 'IDLE', COUNT(*), ROUND(SUM(LOSS_MINUTES)/60,1), 0, 0,
       ROUND(SUM(FORGONE_MARGIN),0)
FROM SNOWCORE_REAL.GOLD.PRODUCTION_LOSS_INCIDENT;

-- G. Work-order priority mix
SELECT PRIORITY, COUNT(*) ORDERS FROM SNOWCORE_REAL.GOLD.WORK_ORDER
WHERE PLANT_CODE='PLANT_B' GROUP BY PRIORITY ORDER BY PRIORITY;

-- H. Stock states
SELECT STOCK_STATE, COUNT(*) MATERIALS FROM SNOWCORE_REAL.GOLD.INVENTORY_SNAPSHOT
GROUP BY STOCK_STATE ORDER BY STOCK_STATE;

-- I. Downtime event durations (breakdown vs idle), minutes
SELECT STOP_STATE, COUNT(*) EVENTS,
       ROUND(MEDIAN(STOP_DURATION_MIN),2) MEDIAN_MIN,
       ROUND(APPROX_PERCENTILE(STOP_DURATION_MIN,0.9),1) P90_MIN,
       ROUND(SUM(STOP_DURATION_MIN)/60,1) HOURS
FROM SNOWCORE_REAL.GOLD.PIADE_DOWNTIME_EVENT GROUP BY STOP_STATE;

-- J. Hourly target base rate by month (is the next hour >10% downtime)
SELECT TO_CHAR(DATE_TRUNC('month',HOUR_TS),'YYYY-MM') MONTH,
       COUNT(*) HOURS, ROUND(AVG(LABEL_HEAVY_STOP),4) HEAVY_STOP_RATE
FROM SNOWCORE_REAL.ML.PLANT_B_FEATURES GROUP BY 1 ORDER BY 1;
```
