# Cortex prompt 042 — read-only facts for the A–Z project guide

Use connection `hackathon`. Read-only: SELECT only. Run each query as its own
call. Print every result as a markdown table with ALL rows and ALL columns
verbatim. Do not summarise. If a query fails, print the exact error and continue.

```sql
SELECT MACHINE_CODE, ROUND(IDEAL_RATE_PPH,0) IDEAL, ROUND(P99_SPEED_PPH,0) P99, ROUND(MEDIAN_SPEED_PPH,0) MEDIAN_SPEED
FROM SNOWCORE_REAL.GOLD.OEE_IDEAL_RATE ORDER BY MACHINE_CODE;

SELECT GRAIN, SUBJECT, LOSS_BUCKET, OWNER_FUNCTION, ROUND(LOSS_HOURS,0) LOSS_HOURS,
       ROUND(FORGONE_UNITS,0) FORGONE_UNITS, ROUND(PCT_OF_SUBJECT_FORGONE,1) PCT_OF_SUBJECT,
       ROUND(SCENARIO_SIGNED_LOSS_MARGIN_EUR,0) MARGIN_EUR
FROM SNOWCORE_REAL.GOLD.V_OEE_LOSS_ATTRIBUTION WHERE GRAIN='SITE' ORDER BY LOSS_BUCKET;

SELECT LEVER_RANK, MACHINE_CODE, LOSS_BUCKET, ROUND(OBSERVED_LOSS_HOURS,0) HOURS,
       ROUND(OBSERVED_FORGONE_PACKAGES,0) PACKAGES, ROUND(PCT_OF_SITE_BUCKET_LOSS,1) PCT_SITE_BUCKET,
       ROUND(SCENARIO_SIGNED_LOSS_MARGIN_EUR,0) MARGIN_EUR, TOP_CONTRIBUTING_CAUSE
FROM SNOWCORE_REAL.GOLD.V_IMPROVEMENT_LEVERS ORDER BY LEVER_RANK LIMIT 6;

SELECT ACTION_RANK, PRIORITY, OWNER_FUNCTION, MACHINE_CODE, RISK_STATE,
       ROUND(PRODUCTION_EXPOSURE_EUR,0) EXPOSURE_EUR
FROM SNOWCORE_REAL.GOLD.V_EXECUTIVE_ACTION_QUEUE ORDER BY ACTION_RANK;

SELECT MACHINE_CODE, WORK_ORDERS, ROUND(BREAKDOWN_HOURS,0) BD_H, ROUND(LABOUR_COST,0) LABOUR,
       ROUND(PARTS_COST,0) PARTS, ROUND(BREAKDOWN_FORGONE_MARGIN,0) BD_MARGIN,
       ROUND(MAINTENANCE_COST,0) MAINT, IDLE_INCIDENTS, ROUND(IDLE_HOURS,0) IDLE_H,
       ROUND(IDLE_FORGONE_MARGIN,0) IDLE_MARGIN, ROUND(PCT_COST_FROM_IDLING,1) PCT_IDLE
FROM SNOWCORE_REAL.GOLD.V_PIADE_COST_BY_MACHINE ORDER BY MACHINE_CODE;

SELECT MACHINE_CODE, ROUND(SUM(MARGIN_EXPOSURE_EUR),0) EXPOSURE_EUR, COUNT(*) ORDERS,
       COUNT_IF(ORDER_STATUS='TARGET_MET') MET, COUNT_IF(ORDER_STATUS='BELOW_TARGET') BELOW,
       COUNT_IF(ORDER_STATUS='AT_RISK') AT_RISK
FROM SNOWCORE_REAL.GOLD.PRODUCTION_ORDER GROUP BY MACHINE_CODE ORDER BY MACHINE_CODE;

SELECT FEATURE, ROUND(IMPORTANCE,4) IMPORTANCE FROM SNOWCORE_REAL.ML.PLANT_B_FEATURE_IMPORTANCE ORDER BY RANK LIMIT 8;

SELECT RISK_BAND, HOURS, ROUND(MEAN_SCORE,3) MEAN_SCORE, ROUND(ACTUAL_RATE,3) ACTUAL_RATE
FROM SNOWCORE_REAL.ML.PLANT_B_CALIBRATION_BAND ORDER BY MEAN_SCORE;

SELECT SCOPE, TEST_ROWS, ROUND(BASE_RATE,3) BASE_RATE, ROUND(AUC,4) AUC,
       ROUND(TOP_DECILE_PRECISION,4) P, ROUND(BASELINE_TOP_DECILE_PRECISION,4) BP,
       ROUND(TOP_DECILE_RECALL,4) R, TRAIN_ROWS
FROM SNOWCORE_REAL.ML.PLANT_B_MODEL_METRICS ORDER BY SCOPE;

SELECT CAMPAIGN_ID, STATUS, TRIALS_COMPLETED, TRIALS_REJECTED, BASELINE_MEAN_AUC,
       CHAMPION_MEAN_AUC, PROMOTION_STATUS
FROM SNOWCORE_REAL.ML.PIADE_RESEARCH_CAMPAIGN ORDER BY STARTED_AT DESC LIMIT 3;

SELECT TRIAL_NUMBER, MODEL_FAMILY, TRIAL_DECISION, ROUND(MEAN_AUC,4) MEAN_AUC,
       ROUND(WORST_AUC,4) WORST_AUC, IS_CHAMPION
FROM SNOWCORE_REAL.ML.V_PIADE_RESEARCH_TRIALS
WHERE CAMPAIGN_ID LIKE '233717d9%' ORDER BY TRIAL_NUMBER;

SELECT 'BRONZE.PIADE_INTERVALS_RAW' T, COUNT(*) N FROM SNOWCORE_REAL.BRONZE.PIADE_INTERVALS_RAW
UNION ALL SELECT 'BRONZE.PIADE_HOURLY_RAW', COUNT(*) FROM SNOWCORE_REAL.BRONZE.PIADE_HOURLY_RAW
UNION ALL SELECT 'SILVER.PIADE_INTERVAL', COUNT(*) FROM SNOWCORE_REAL.SILVER.PIADE_INTERVAL
UNION ALL SELECT 'GOLD.PIADE_INTERVAL_SEGMENT', COUNT(*) FROM SNOWCORE_REAL.GOLD.PIADE_INTERVAL_SEGMENT
UNION ALL SELECT 'GOLD.PIADE_OEE_HOURLY', COUNT(*) FROM SNOWCORE_REAL.GOLD.PIADE_OEE_HOURLY
UNION ALL SELECT 'GOLD.PIADE_OEE_DAILY', COUNT(*) FROM SNOWCORE_REAL.GOLD.PIADE_OEE_DAILY
UNION ALL SELECT 'GOLD.PIADE_DOWNTIME_EVENT', COUNT(*) FROM SNOWCORE_REAL.GOLD.PIADE_DOWNTIME_EVENT
UNION ALL SELECT 'ML.PLANT_B_FEATURES', COUNT(*) FROM SNOWCORE_REAL.ML.PLANT_B_FEATURES
UNION ALL SELECT 'GOLD.V_ANALYST_DECISION_CONTEXT', COUNT(*) FROM SNOWCORE_REAL.GOLD.V_ANALYST_DECISION_CONTEXT
UNION ALL SELECT 'GOLD.WORK_ORDER PLANT_A', COUNT(*) FROM SNOWCORE_REAL.GOLD.WORK_ORDER WHERE PLANT_CODE='PLANT_A';

SELECT MACHINE_CODE, TO_TIMESTAMP_NTZ(HOUR_TS/1000000000) LAST_HOUR, ROUND(RISK_SCORE,4) SCORE, RISK_BAND
FROM SNOWCORE_REAL.ML.PLANT_B_RISK_SCORE
QUALIFY ROW_NUMBER() OVER (PARTITION BY MACHINE_CODE ORDER BY HOUR_TS DESC)=1 ORDER BY MACHINE_CODE;
```
