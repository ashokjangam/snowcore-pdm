```sql
SELECT SPLIT_PART, COUNT(*) ROWS, COUNT_IF(FUTURE_HOURS_OBSERVED=4) COMPLETE_ROWS,
       AVG(IFF(FUTURE_HOURS_OBSERVED=4, LABEL_BREAKDOWN_4H, NULL)) POSITIVE_RATE,
       MIN(HOUR_TS) FIRST_HOUR, MAX(HOUR_TS) LAST_HOUR
FROM SNOWCORE_REAL.ML.PLANT_B_FAILURE_LABEL GROUP BY SPLIT_PART ORDER BY FIRST_HOUR;

SELECT SCOPE, TEST_ROWS, BASE_RATE, MODEL_AUC, PERSISTENCE_AUC, RECENT_AUC,
       MODEL_TOP_DECILE_PRECISION, PERSISTENCE_TOP_DECILE_PRECISION,
       RECENT_TOP_DECILE_PRECISION, MODEL_TOP_DECILE_RECALL,
       BEST_BASELINE, MODEL_ADVANTAGE_PTS, VERDICT, TRAIN_ROWS,
       EXCLUDED_INCOMPLETE_ROWS, FLAG_THRESHOLD_SCORE
FROM SNOWCORE_REAL.ML.PLANT_B_FAILURE_METRICS ORDER BY SCOPE;

SELECT FEATURE, IMPORTANCE FROM SNOWCORE_REAL.ML.PLANT_B_FAILURE_IMPORTANCE
ORDER BY RANK LIMIT 10;

SELECT COUNT(*) SCORED, COUNT_IF(HOUR_TS < '2021-12-01'::TIMESTAMP_NTZ) NON_BLIND,
       SUM(IS_FLAGGED) FLAGGED, CEIL(COUNT(*)*0.10) EXPECTED_FLAGGED
FROM SNOWCORE_REAL.ML.PLANT_B_FAILURE_SCORE;

/* Alignment check: long-breakdown minutes seen by the label should fall in
   hours whose published PCT_DOWNTIME is non-zero. */
SELECT COUNT(*) LONG_EVENTS,
       COUNT_IF(h.PCT_DOWNTIME > 0) IN_HOUR_WITH_DOWNTIME
FROM SNOWCORE_REAL.GOLD.PIADE_DOWNTIME_EVENT e
JOIN SNOWCORE_REAL.BRONZE.PIADE_HOURLY_RAW h
  ON 'PLANT_B:'||h.EQUIPMENT_ID = e.MACHINE_KEY
 AND h.INTERVAL_START = DATE_TRUNC('hour', CONVERT_TIMEZONE('UTC', e.STOP_START)::TIMESTAMP_NTZ)
WHERE e.STOP_STATE='downtime' AND e.STOP_DURATION_MIN >= 10;

SELECT MACHINE_CODE, COUNT(*) CODES, SUM(LONG_BREAKDOWNS) LONG_BREAKDOWNS,
       ROUND(SUM(BREAKDOWN_HOURS),1) HOURS,
       MAX(IFF(RANK_IN_LINE=1, ALARM_CODE, NULL)) TOP_CODE,
       ROUND(MAX(IFF(RANK_IN_LINE=1, SHARE_OF_LINE_BREAKDOWN_PCT, NULL)),1) TOP_SHARE_PCT,
       MAX(IFF(RANK_IN_LINE=1, PATTERN, NULL)) TOP_PATTERN
FROM SNOWCORE_REAL.GOLD.V_ROOT_CAUSE_ALARM GROUP BY MACHINE_CODE ORDER BY MACHINE_CODE;

SELECT * FROM SNOWCORE_REAL.GOLD.V_RISK_WORK_ORDER_SUMMARY ORDER BY SCOPE;

SELECT RISK_WORK_ORDER_ID, MACHINE_CODE, ISSUED_AT, PRIORITY, EPISODE_FLAGGED_HOURS,
       TOP_RECENT_ALARM, STOCK_STATE, OUTCOME, LEAD_TIME_MIN
FROM SNOWCORE_REAL.GOLD.RISK_WORK_ORDER ORDER BY ISSUED_AT LIMIT 8;

SELECT COUNT(*) RISK_TABLE_ROWS FROM SNOWCORE_REAL.ML.PLANT_B_RISK_SCORE;
```
